<%:Service Status%>

<%:Bypass Mode%>
<% if enabled then %> <%:Enabled%> <% else %> <%:Disabled%> <% end %>
<%:LAN IPv4 Address%>
<%=pcdata(lan_ip ~= "" and lan_ip or "N/A")%>
<%:LAN IPv6 Address%>
<%=pcdata(lan_ip6 ~= "" and lan_ip6 or "N/A")%>
<%:IPv4 Gateway%>
<%=pcdata(gateway ~= "" and gateway or "N/A")%>
<%:IPv6 Gateway%>
<%=pcdata(gateway6 ~= "" and gateway6 or "N/A")%>
<%:IPv6 Mode%>
<% if ipv6_mode == "nat6" then %> NAT6 <% elseif ipv6_mode == "relay" then %> Relay <% elseif ipv6_mode == "hybrid" then %> Hybrid <% else %> <%:Disabled%> <% end %>
<%:IPv6 Forwarding%>
<% if ipv6_forward then %> <%:Enabled%> <% else %> <%:Disabled%> <% end %>
<%:NAT6 Status%>
<% if nat6_active then %> <%:Active%> (<%=nat6_count%> <%:rules%>) <% else %> <%:Inactive%> <% end %>

<% if ipv4_conn then %> ✓ <%:Connected%> <% else %> ✗ <%:Disconnected%> <% end %>
